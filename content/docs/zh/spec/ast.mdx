## 概述

本文档定义了 XUTO 程序的抽象语法树（AST）表示。规范采用 TypeScript 风格符号描述，但与语言无关。

## 节点基础

所有 AST 节点共享公共基础：

```typescript
interface BaseNode {
  type: string;
  span: { start: number; end: number };
}
```

## 程序

```typescript
interface Program extends BaseNode {
  type: 'Program';
  body: Declaration[];
}
```

## 声明

```typescript
interface FnDeclaration extends BaseNode {
  type: 'FnDeclaration';
  isExport: boolean;
  name: IdentifierNode;
  params: ParameterNode[];
  returnType: TypeNode | null;
  body: BlockNode;
}

interface StructDeclaration extends BaseNode {
  type: 'StructDeclaration';
  isExport: boolean;
  isPub: boolean;
  name: IdentifierNode;
  genericParams: IdentifierNode[];
  fields: FieldNode[];
}

interface EnumDeclaration extends BaseNode {
  type: 'EnumDeclaration';
  isExport: boolean;
  name: IdentifierNode;
  variants: EnumVariantNode[];
}

interface TraitDeclaration extends BaseNode {
  type: 'TraitDeclaration';
  isExport: boolean;
  name: IdentifierNode;
  methods: TraitMethodNode[];
}

interface ImplDeclaration extends BaseNode {
  type: 'ImplDeclaration';
  traitName: IdentifierNode | null;
  targetType: TypeNode;
  methods: FnDeclaration[];
}

interface TypeAlias extends BaseNode {
  type: 'TypeAlias';
  isExport: boolean;
  name: IdentifierNode;
  genericParams: IdentifierNode[];
  initType: TypeNode;
}
```

## 参数与字段

```typescript
interface ParameterNode extends BaseNode {
  type: 'Parameter';
  name: IdentifierNode;
  paramType: TypeNode;
  isOptional: boolean;
  default: ExpressionNode | null;
}

interface FieldNode extends BaseNode {
  type: 'Field';
  isPub: boolean;
  name: IdentifierNode;
  fieldType: TypeNode;
}

interface EnumVariantNode extends BaseNode {
  type: 'EnumVariant';
  name: IdentifierNode;
  payload: TypeNode[];
}
```

## 类型

```typescript
type TypeNode =
  | IdentifierType
  | GenericType
  | OptionalType
  | UnionType
  | IntersectionType
  | FnType
  | ObjectType;

interface IdentifierType extends BaseNode {
  type: 'IdentifierType';
  name: string;
}

interface GenericType extends BaseNode {
  type: 'GenericType';
  base: TypeNode;
  args: TypeNode[];
}

interface OptionalType extends BaseNode {
  type: 'OptionalType';
  inner: TypeNode;
}

interface UnionType extends BaseNode {
  type: 'UnionType';
  left: TypeNode;
  right: TypeNode;
}

interface IntersectionType extends BaseNode {
  type: 'IntersectionType';
  left: TypeNode;
  right: TypeNode;
}

interface FnType extends BaseNode {
  type: 'FnType';
  params: TypeNode[];
  returnType: TypeNode | null;
}
```

## 表达式

```typescript
type ExpressionNode =
  | IdentifierNode
  | NumberLiteral
  | StringLiteral
  | BooleanLiteral
  | NullLiteral
  | BinaryExpr
  | UnaryExpr
  | CallExpr
  | MemberExpr
  | IndexExpr
  | TernaryExpr
  | IfExpr
  | MatchExpr
  | FnExpr
  | ListExpr
  | ObjectExpr
  | ComponentExpr;

interface BinaryExpr extends BaseNode {
  type: 'BinaryExpr';
  op: string;
  left: ExpressionNode;
  right: ExpressionNode;
}

interface CallExpr extends BaseNode {
  type: 'CallExpr';
  callee: ExpressionNode;
  args: ExpressionNode[];
}

interface ComponentExpr extends BaseNode {
  type: 'ComponentExpr';
  name: IdentifierNode;
  args: NamedArgNode[];
  body: BlockNode | null;
}

interface MatchExpr extends BaseNode {
  type: 'MatchExpr';
  subject: ExpressionNode;
  arms: MatchArmNode[];
}
```

## 语句

```typescript
type StatementNode =
  | VariableDecl
  | ReturnStatement
  | ThrowStatement
  | ExpressionStatement;

interface VariableDecl extends BaseNode {
  type: 'VariableDecl';
  isMutable: boolean;
  name: IdentifierNode;
  varType: TypeNode | null;
  init: ExpressionNode;
}
```

## 公共节点

```typescript
interface IdentifierNode extends BaseNode {
  type: 'Identifier';
  value: string;
}

interface BlockNode extends BaseNode {
  type: 'Block';
  body: StatementNode[];
}

interface NamedArgNode extends BaseNode {
  type: 'NamedArg';
  name: IdentifierNode;
  value: ExpressionNode;
}
```

## 序列化格式

AST 节点**必须**可序列化为 JSON。`span` 字段记录源文件中基于零的字符偏移量 `(start, end)`。
